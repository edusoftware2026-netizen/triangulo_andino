// lib/geminiService.ts
import { GoogleGenerativeAI } from "@google/generative-ai";
import { APP_CONFIG } from "./config";

export async function generateItineraryPlan(payload: {
  destinations: string[];
  days: number;
  travelers: number;
  category: string;
}) {
  if (!APP_CONFIG.useRealGeminiAI) {
    console.log("⚡ Dual Mode: Despachando Mock Engine local...");
    return getLocalOptimizedMock(payload);
  }

  // Llamada Real al SDK de Google Gemini con Grounding de búsqueda
  const genAI = new GoogleGenerativeAI(process.env.GEMINI_API_KEY!);
  const model = genAI.getGenerativeModel({
    model: "gemini-1.5-pro",
    tools: [{ googleSearch: {} }] // Google Search Grounding habilitado
  });

  const prompt = `Actúa como Concierge de Lujo de Triángulo Andino. Diseña un circuito para ${payload.days} días, 
  destinos: ${payload.destinations.join(", ")}, con altitud gradual para evitar soroche.`;

  const result = await model.generateContent(prompt);
  return result.response.text();
}
// lib/firebase.ts
import { initializeApp, getApps } from "firebase/app";
import { getFirestore, doc, setDoc } from "firebase/firestore";
import { APP_CONFIG } from "./config";

const firebaseConfig = {
  apiKey: process.env.NEXT_PUBLIC_FIREBASE_API_KEY,
  authDomain: process.env.NEXT_PUBLIC_FIREBASE_AUTH_DOMAIN,
  projectId: process.env.NEXT_PUBLIC_FIREBASE_PROJECT_ID,
  storageBucket: process.env.NEXT_PUBLIC_FIREBASE_STORAGE_BUCKET,
};

export const app = !getApps().length && APP_CONFIG.useRealFirebase 
  ? initializeApp(firebaseConfig) 
  : null;

export const db = app ? getFirestore(app) : null;

// Guardado de cotizaciones con fallback instantáneo a localStorage
export async function persistQuote(quoteData: Record<string, any>) {
  if (db) {
    const quoteRef = doc(db, "quotes", quoteData.id);
    return await setDoc(quoteRef, quoteData);
  } else {
    const existing = JSON.parse(localStorage.getItem("triangulo_quotes") || "[]");
    existing.push({ ...quoteData, savedAt: new Date().toISOString(), mode: "local_mock" });
    localStorage.setItem("triangulo_quotes", JSON.stringify(existing));
    return true;
  }
}
